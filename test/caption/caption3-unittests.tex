\newcommand*\passed[1]{%
  \typeout{Test `#1' passed.}}
\newcommand*\failed[1]{%
  \PackageError{caption3}{Test `#1' failed}{}}

\newcommand*\test@defined[2]{%
  \@ifundefined{#2}\failed\passed{#1}}
\newcommand*\test@undefined[2]{%
  \@ifundefined{#2}\passed\failed{#1}}
\newcommand*\test@equal[3]{%
  \begingroup
    \def\test@temp{#3}%
    \ifx#2\test@temp\expandafter\passed\else\expandafter\failed\fi{#1}%
  \endgroup}
\newcommand*\test@ident[3]{%
  \ifx#2#3\expandafter\passed\else\expandafter\failed\fi{#1}}
