\usepackage[figureposition=b,tableposition=t]{caption}
\usepackage[list=on]{subcaption}

\DeclareCaptionSubType*{figure}
\renewcommand\thesubfigure{\thefigure\alph{subfigure}}
\DeclareCaptionSubType*{table}
\renewcommand\thesubtable{\thetable.\arabic{subtable}}

\newcommand\makesubfigure[1]{%
  \begin{subfigure}{.4\textwidth}\centering\includegraphics[width=.5\hsize,height=.5\hsize]{demo}\caption{#1}\end{subfigure}}
\newcommand\makesubtable[1]{%
  \begin{subtable}{.4\textwidth}\caption{#1}\centering\begin{tabular}{cc}A & B\\ C & D\\ \end{tabular}\end{subtable}}
