\usepackage{hyperref}
\usepackage{caption}
\captionsetup[table]{labelfont=bf}

\begin{document}

\begin{table}
  \caption{Some test caption for a regular table}
  \centering
  \begin{tabular}{ll}
    Foo & Baz \\
    baz & foobaz \\
  \end{tabular}
\end{table}

\end{document}
