% Tagged PDF: longtable captions (caption v3.7 together with latex-lab's table tagging)
\DocumentMetadata{lang=en-US,tagging=on}
\documentclass{article}
\usepackage{longtable}
\usepackage[labelfont=bf,labelsep=period]{caption}
\begin{document}
\listoftables

\begin{longtable}{ll}
  \caption{A long table}\label{tab:a}\\
  A & B \\
  C & D \\
\end{longtable}

\begin{longtable}{ll}
  \caption*{An unnumbered long table}\\
  A & B \\
\end{longtable}

\begin{table}[h]
  \centering
  \caption{A table}\label{tab:b}
  \begin{tabular}{ll} A & B \end{tabular}
\end{table}

References: \ref{tab:a}, \ref{tab:b}.
\end{document}
