% Proposal (c), item 4: caption.sty 3.6 under tagging keeps its own
% \@makecaption when latex-lab's begindocument code is removed by its label
\DocumentMetadata{tagging=on}
\documentclass{article}
\usepackage[labelformat=parens,labelsep=period,labelfont=bf,format=hang]{caption}
\RemoveFromHook{begindocument}[latex-lab/float/makecaption]
\begin{document}
\begin{figure}[h]
  \caption{A caption that is long enough to need more than one line, so that the
    hanging indentation of format=hang becomes visible in the output}
\end{figure}
\end{document}
