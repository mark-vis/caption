% a1: the generic command hook cmd/@xfloat/after as an alternative to wrapping
% \@xfloat (what caption needs: run code inside the float box, after \@captype
% is set). Works in a plain document, fails with babel-french.
\documentclass{article}
\usepackage[T1]{fontenc}
\usepackage[french]{babel}
\makeatletter
\AddToHookWithArguments{cmd/@xfloat/after}{%
  \typeout{FLOAT-BEGIN: type=\@captype, hsize=\the\hsize}}
\makeatother
\begin{document}
\begin{figure}[ht]
\centering X
\caption{Une figure}
\end{figure}
\end{document}
