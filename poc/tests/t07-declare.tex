% \DeclareCaption... user declarations
\documentclass{article}
\usepackage{%CAPTION%}
\DeclareCaptionFormat{myformat}{#1#2\textbf{[}#3\textbf{]}}
\DeclareCaptionLabelFormat{mylabel}{\textsc{#1} (#2)}
\DeclareCaptionLabelSeparator{arrow}{ $\rightarrow$ }
\DeclareCaptionTextFormat{up}{\MakeUppercase{#1}}
\DeclareCaptionFont{red}{\itshape}
\captionsetup{format=myformat,labelformat=mylabel,labelsep=arrow,textformat=up,textfont=red}
\begin{document}
\begin{figure}[h]\centering\rule{2cm}{1cm}\caption{Declared things}\end{figure}
\begin{figure}[h]\centering\rule{2cm}{1cm}\caption[]{}\end{figure}
\end{document}
