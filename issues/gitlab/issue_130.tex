\newcommand\testcounter[2]{%
  \edef\testa{\arabic{#1}}%
  \edef\testb{#2}%
  \ifx\testa\testb
    \typeout{Test #1:#2 passed}%
  \else
    \typeout{*** Test #1:#2 failed}%
    \show\testa
  \fi}
