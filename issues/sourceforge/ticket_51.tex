% \DeclareCaptionLabelFormat does not behave as expected when used with floating environment defined by the float package:

\documentclass{article}
\usepackage{algorithmic}
\usepackage{algorithm}

\usepackage{caption}
\DeclareCaptionLabelFormat{unnumbered}{#1}
%\makeatletter
%\DeclareCaptionLabelFormat{unnumbered}{\fname@algorithm}
%\makeatother

\begin{document}

%\listofalgorithms

\begin{algorithm}[H]
\caption{great}
\begin{algorithmic}
\STATE dummy
\end{algorithmic}
\end{algorithm}

\begin{figure}[H]
\caption{great}
\end{figure}

\captionsetup{labelformat=unnumbered,listformat=empty}
  % does not have any effect

\begin{algorithm}[H]
\caption{great}
\begin{algorithmic}
\STATE dummy
\end{algorithmic}
\end{algorithm}

\begin{figure}[H]
\caption{great}
\end{figure}

\captionsetup[algorithm]{labelformat=unnumbered,listformat=empty}
 % makes "Algorithm" disappear

\begin{algorithm}[H]
\caption{great}
\begin{algorithmic}
\STATE dummy
\end{algorithmic}
\end{algorithm}

\begin{figure}[H]
\caption{great}
\end{figure}

\end{document}
