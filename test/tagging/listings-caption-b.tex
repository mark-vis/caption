% Captions of the listings package with \DocumentMetadata{} (without tagging)
% The caption package is loaded before the listings package.
%
% With LaTeX 2026-11-01 the tagging support of the listings package
% (latex-lab-listings) re-defines \lst@makecaption and \lst@maketitle.
% caption v3.6p replaced these definitions by its own; without tagging
% this gave no errors, but title= was typeset as ": A title".
%
% If latex-lab-listings.sty is not available (LaTeX before 2026-11-01, where
% the listings package is not adapted), the document is compiled without
% \DocumentMetadata.

\IfFileExists{latex-lab-listings.sty}{\DocumentMetadata{}}{}
\documentclass{article}
\usepackage{caption}
\usepackage{listings}

\begin{document}

\lstlistoflistings

\section{Listings}

\begin{lstlisting}[caption={A listing},label=lst:a]
abc
\end{lstlisting}

\begin{lstlisting}[title={A title}]
def
\end{lstlisting}

\begin{lstlisting}[float=htbp,caption={A floating listing},label=lst:b]
ghi
\end{lstlisting}

\begin{lstlisting}[float=htbp,title={A floating title}]
jkl
\end{lstlisting}

See listing~\ref{lst:a} and listing~\ref{lst:b}.

\end{document}

% Expected output:
% No errors. "Listing 1: A listing", "A title" (without label and without
% a leading colon), "Listing 2: A floating listing" and "A floating title";
% the list of listings contains the two numbered listings, the references
% give "1" and "2".
% The output is the same as without the caption package.
% (With \DocumentMetadata the captions are typeset by the LaTeX kernel, so
% caption package options like font= do not apply to them.)
