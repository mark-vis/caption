% Issue: 125
% Title: \captionsetup*{type=...} does not suppress the warning "The type was already set to ..."
% Date:  Apr 1, 2021
% URL:   https://github.com/mark-vis/caption/issues/28 (moved from https://gitlab.com/axelsommerfeldt/caption/issues/125)

% The documentation (section A.3 "Warnings") states that the warning
% "The caption type was already set to <type>" can be suppressed by using
% the starred form \captionsetup*{type=...}. Since caption3 v2.4 the star
% was ignored here, so the warning was issued anyway.

\documentclass{article}

\usepackage{caption}
\captionsetup[figure]{labelfont=bf}
\captionsetup[table]{textfont=bf}

\begin{document}

\begin{figure}
  \caption{Figure caption}
  \captionsetup*{type=table}% no warning
  \caption{Table caption}
\end{figure}

\begin{figure}
  \caption{Figure caption}
  \captionsetup{type=table}% warning
  \caption{Table caption}
\end{figure}

\end{document}

% Expected output:
% "Figure 1: Figure caption" and "Table 1: Table caption" (the latter in bold),
% "Figure 2: Figure caption" and "Table 2: Table caption" (the latter in bold),
% and exactly one warning "The type was already set to `figure'" (for line 27).
