% Proposal (c): caption3's label settings through latex-lab's sockets
\DocumentMetadata{tagging=on}
\documentclass{article}
\usepackage{caption3}
\makeatletter\input{caption3-labelplug.tex}\makeatother
\captionsetup{labelformat=parens,labelsep=period,labelfont=bf}
\AssignSocketPlug{caption/label}{caption3}
\AssignSocketPlug{caption/separator}{caption3}
\begin{document}
\begin{figure}[h]
  \caption{Short caption}
\end{figure}
\begin{figure}[h]
  \caption{A long caption that does not fit on a single line, so the multi-line
    branch of latex-lab's \texttt{\string\@makecaption} is used here as well}
\end{figure}
\end{document}
