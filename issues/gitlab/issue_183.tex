% Since KOMA-Script 3.46 tocbasic has its own implementation of the caption
% position which supports caption via \captionsetup, and the classes no longer
% define \if@tablecaptionabove or \@setcapindent. With caption-koma.sto v2.0e
% loading caption-koma.sto with such a class gave 7 errors (\@setcapindent,
% \@@setcapindent, \@tablecaptionabovetrue ...), and its position patches
% overrode the ones of tocbasic.
%
% KOMA-Script 3.49 prevents loading of caption-koma.sto by setting
% \caption@documentclass to `standard' in the package/caption3/before hook.
% This test switches that off, so caption-koma.sto IS loaded.

\documentclass[captions=tableheading]{scrartcl}

\makeatletter
\AddToHook{package/caption3/before}[issue183]{\def\caption@documentclass{koma}}
\DeclareHookRule{package/caption3/before}{issue183}{after}{tocbasic}
\makeatother

\usepackage{longtable}
\usepackage[tableposition=top]{caption}
\setcapindent{1em}

\begin{document}

\begin{figure}
  \centering
  \rule{3cm}{1cm}
  \caption{Figure caption}
\end{figure}

\begin{table}
  \caption{Table caption}
  \centering
  x
\end{table}

\begin{longtable}{l}
  \captionabove{Longtable caption}\\
  x\\
\end{longtable}

\end{document}
