% Since amsmath 2025-05-18 \@arrayparboxrestore resets \label to the kernel
% definition, so the second label `sub@...' was no longer written inside
% \subcaptionbox and \subref printed `??'.
% Expected output: a b a and 1a 1b 1a

\documentclass{article}
\usepackage{amsmath}
\usepackage{graphicx}
\usepackage{subcaption}
\usepackage{hyperref}
\usepackage{cleveref}

\begin{document}

\begin{figure}
\centering
\subcaptionbox{A\label{fig:a}}{\includegraphics[width=0.3\textwidth]{example-image-a}}
\begin{subfigure}{0.3\textwidth}
  \includegraphics[width=\linewidth]{example-image-b}
  \caption{B}\label{fig:b}
\end{subfigure}
\caption{Main}\label{fig:main}
\end{figure}

\begin{table}
\centering
\begin{subtable}{0.3\textwidth}
  \centering
  \begin{tabular}{cc} 1 & 2 \end{tabular}
  \caption{T}\label{tab:a}
\end{subtable}
\caption{Main}\label{tab:main}
\end{table}

\subref{fig:a} \subref{fig:b} \subref{tab:a} and \ref{fig:a} \ref{fig:b} \ref{tab:a}

\end{document}
