% Tagged PDF: longtable captions (caption v3.7, ltcaption v1.5)
% together with latex-lab's table tagging
\DocumentMetadata{lang=en-US,tagging=on}
\documentclass{article}
\usepackage{longtable}
\usepackage{ltcaption}
\usepackage[labelfont=bf,labelsep=period,font=small]{caption}
\begin{document}
\listoftables

\begin{longtable}{ll}
  \caption{A long table}\label{tab:a}\\
  A & B \\
  C & D \\
\end{longtable}

\begin{longtable}{ll}
  \caption*{An unnumbered long table}\\
  A & B \\
\end{longtable}

\begin{longtable}{ll}
  \caption[]{A long table without list entry}\label{tab:c}\\
  A & B \\
\end{longtable}

\begin{longtable}{ll}
  \caption[Short]{A long table with a very long caption text which needs more than
    one line, so it will be typeset as a paragraph with several lines}\label{tab:d}\\
  A & B \\
\end{longtable}

\begin{longtable*}{ll}
  \caption*{A longtable* caption}\\
  A & B \\
\end{longtable*}

\begin{longtable}{ll}
  \captionsetup{labelformat=empty}
  \caption{Empty label format}\\
  A & B \\
\end{longtable}

\begin{longtable}{ll}
  A & B \\
  \caption{Caption at the bottom}\label{tab:g}
\end{longtable}

\begin{table}[h]
  \centering
  \caption{A table}\label{tab:h}
  \begin{tabular}{ll} A & B \end{tabular}
\end{table}

References: \ref{tab:a}, \ref{tab:c}, \ref{tab:d}, \ref{tab:g}, \ref{tab:h}.
\end{document}
