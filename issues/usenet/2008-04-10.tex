\documentclass{scrbook}
\usepackage{amssymb}
\usepackage{calc}
\usepackage{caption}
\usepackage{float}
\usepackage{graphicx}

 \floatstyle{plain}
 \newfloat{beispielfloat}{htbp}{lob}

 \newlength{\ITCVbeispiellena}
 \setlength{\ITCVbeispiellena}{\widthof{$\vartriangleright$}}
 \newenvironment{test}[3][htb]{%
    \begin{beispielfloat}[#1]
    \noindent$\vartriangleright$%
    \rule[0.535ex]{\linewidth-2\ITCVbeispiellena}{0.35pt}%
    $\vartriangleleft$%
    \par
    \noindent Beispiel 1%
 }{%
    \noindent$\vartriangleright$%
    \rule[0.535ex]{\linewidth-2\ITCVbeispiellena}{0.35pt}%
    $\vartriangleleft$%
    \end{beispielfloat}
 }

\begin{document}
 \begin{test}{}{}

  Test

  \begin{figure}[H]
  A
  \caption{Test}
  \end{figure}

  Test

 \end{test}
\end{document}
