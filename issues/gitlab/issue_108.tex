\documentclass[a4paper]{article}
\usepackage{graphicx}

% Use "ngerman" as 1st language, "english" as 2nd one
\usepackage[english,ngerman]{babel}

% Load the bicaption package with 2nd language set to
% "english", and list type "figureEng" resp. "tableEng"
\usepackage[lang=english,listtype+=Eng,font=it]{bicaption}
\captionsetup{slc=off} % do not center short captions

\usepackage{listings}

% Change English name of "Listing" to "Code"
\addto\captionsenglish{\renewcommand\lstlistingname{Code}}

\usepackage{newfloat}
% Define the new floating environment type "figureEng"
% (just to get an extra list for English figure captions)
\DeclareFloatingEnvironment[fileext=lof2]{figureEng}[Figure][List of Figures]
% Define the new floating environment type "lstlistingEng"
% (just to get an extra list for English listing captions)
\DeclareFloatingEnvironment[fileext=lol2]{lstlistingEng}[Code][List of Codes]

\begin{document}

\listoffigures        % German
\listoffigureEng      % English

\lstlistoflistings    % German
\listoflstlistingEng  % English

\clearpage

\begin{figure}[!htb]
  \includegraphics[width=4cm]{example-image-a}
  \bicaption[Listeneintrag]{Deutscher Titel}[List entry]{English title}
\end{figure}

\begin{lstlisting}[
  basicstyle=\ttfamily,captionpos=b,language=C,gobble=2,
  caption1={[Listeneintrag]Deutscher Titel}, caption2={[List entry]English Title}
]
  int main()
  {
      printf( "Hello world!\n" );
      return 0;
  }
\end{lstlisting}

\clearpage

% Test listings with option bi-swap
\captionsetup{bi-swap}

\begin{figure}[!htb]
  \includegraphics[width=4cm]{example-image-a}
  \bicaption[List entry]{English title}[Listeneintrag]{Deutscher Titel}
\end{figure}

\begin{lstlisting}[
  basicstyle=\ttfamily,captionpos=b,language=C,gobble=2,
  caption2={[Listeneintrag]Deutscher Titel}, caption1={[List entry]English Title}
]
  int main()
  {
      printf( "Hello world!\n" );
      return 0;
  }
\end{lstlisting}

\end{document}
