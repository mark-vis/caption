\newcommand{\figbox}[1]{%
  \fbox{%                    Frame the box to make the ``figure''
    \vbox to 15mm{%          Make it 15mm tall
    \vfil%                   Vertically center this next \hbox
    \hbox{%                   
      \space
      #1%                    Add the supplied text with spaces
      \space}%
    \vfil}}}
