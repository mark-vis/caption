% Often, floats include the \centering command. With the "newcommands" option of ragged2e, justified captions for such floats are not justified correctly (with or without the caption package). A MWE:

\documentclass{book}
\usepackage{mwe}
\usepackage[newcommands]{ragged2e}
%\usepackage[justification=justified]{caption}
\usepackage[justification=Justified]{caption} % available since caption3.sty v1.9 (2020-05-05)

\begin{document}

\begin{figure}
  \centering
  Not really a figure
  \caption[]{\lipsum[1]}
\end{figure}

\end{document}
