% Within the standard \LaTeX\ document classes captions haven't received the
% attention they deserve. Simply typeset as an ordinary paragraph there is no
% remarkable visual difference from the rest of the text, like here:
%
% \example*{size=normalsize,margin=0pt}{\exampletext}

%
% \example*{size=small,margin=10pt,labelfont=bf,labelsep=endash}{\exampletext}
%
% You can do this easily with this package as there are many ready-to-use

% \section{Option scope}
% \label{optionscope}
%
% \pagebreak[3]
% Options are divided into
% \begin{itemize}
% \item Global options -- Options in the document preamble or later on within the document text
%                         (outside of any \LaTeX\ environment) will change settings globally.
% \item Context-specific options -- Using |\captionsetup|\oarg{context}\marg{options} settings
%                         could be changed for a specific context, for example
%                         |\captionsetup[figure]|\marg{options} will setup options which
%                         are applied \emph{at the beginning} of each |figure| environment,
%                         overriding global ones.
% \item Local options --  Options given within any \LaTeX\ environment are applied immediately
%                         (like global options) but their scope is limited to the current environment.
%                         They override both, global and (usually) context-specific options.
% \end{itemize}
%
% \pagebreak[3]
% For example the document
% \begin{quote}
%   |\documentclass[a4paper]{article}|\\
%   |\usepackage[margin=10pt]{caption}|\\
%   |\captionsetup[figure]{font=it}|\\
%   |\begin{document}|\\
%   |\begin{figure}|\\
%   |  |\ldots\\
%   |  \captionsetup{margin=20pt}|\\
%   |  |\ldots\\
%   |  \caption{A figure}|\\
%   |\end{figure}|\\
%   |\begin{table}|\\
%   |  \caption{A table}|\\
%   |  |\ldots\\
%   |\end{table}|\\
%   |\end{document}|
% \end{quote}
% will typeset the caption ``A figure'' using the settings |margin=20pt| and |font=it|,
% and the caption ``A table'' using the settings |margin=10pt|.
%
% \pagebreak[3]
% Context-specific options are additive, e.g.:
% \begin{quote}
%   |\captionsetup[figure]{labelfont=bf}|\\
%   |\captionsetup[figure]{textfont=it}|
% \end{quote}
% is identical to
% \begin{quote}
%   |\captionsetup[figure]{labelfont=bf,textfont=it}|
% \end{quote}
%
% \pagebreak[3]
% To find out about other contexts please read the section about the use case you are interested in.
% For example there are special contexts for captions typesetted using the \pkg{sidecap}~\cite{sidecap}
% package, so you are able to change settings for these captions only.
% You could also define and use your own ones: Settings specified with
% \begin{quote}
%   |\captionsetup|\oarg{context}\marg{options}
% \end{quote}
% are not applied immediately but stored for the given \meta{context} instead.
% They could be applied later on manually with
% \begin{quote}
%   |\captionsetup{options=|\meta{context}|}|\quad,
% \end{quote}
% for example in own macros (defined via |\new|\-|command|) or environments (defined via |\new|\-|environment|).

% \section{Compatibility}
%
% Unfortunately the \LaTeX\ code used for typesetting captions differs greatly between different
% document classes. To make things even worse, many packages (like \pkg{float}~\cite{float})
% are not using the standard command |\@make|\-|caption| (used by the |\caption| command internally)
% but their very own caption typesetting code instead.
% For this reason the compatibility of this package is limited to those document classes and
% packages it is adapted to.
%
% \pagebreak[3]
% If the \pkg{caption} package has trouble identifing the document class used, it will issue a
% warning:
% \begin{quote}
%   \small
%   |Package caption Warning: Unknown document class (or package),|\\
%   |(caption)                standard defaults will be used.|\\
%   |See the caption package documentation for explanation.|
% \end{quote}
% The usage of this package is \emph{not} recommended in this case, and could cause either
% problems or lead to \LaTeX\ compile errors.

% The \pkg{caption} package also tries to detect the default settings used for captions by the
% detected document class.

% \section{Commands for document class authors}
% \label{class_authors}
%
% A document class could simply load the \pkg{caption} package instead
% of defining an own variant of |\@make|\-|caption|. However if the author decides
% not to do this -- for whatever reason -- there is a second option:
%
% \NEWfeature{v3.5}
% If a document class defines |\caption@document|\-|class|, an adaption to the
% \pkg{caption} package will be loaded as |caption-|\meta{documentclass}|.sto|
% if the \pkg{caption3} kernel is loaded.
% The goal of this adaption file is changing the default settings of the
% \pkg{caption} package so simply loading it (without any package options)
% will not make any harm to the look at feel of the captions.
% This gives end-users the opportunity to fine-tune certain aspects of the
% appearance of captions by adding package options, without unwanted
% side-effects on other aspects of the appearance.
%
% In both cases the command |\Set|\-|Caption|\-|Default|
% could be used to adapt the \pkg{caption} package default values:
%
% \DescribeMacro{\SetCaptionDefault}
% The command
% \begin{quote}
%   |\SetCaptionDefault*|\marg{option}\marg{default value}
% \end{quote}
% changes the default value for the option \meta{option}, for example:
% \begin{quote}
%   |\SetCaptionDefault{format}{hang}|\\
%   |\SetCaptionDefault{justification}{raggedright}|\\
%   |\DeclareCaptionLabelSeparator{silly}{ +++ }|\\
%   |\SetCaptionDefault{labelseparator}{silly}|
% \end{quote}
% This command is available for the options
%   |box|,
%   |font|,
%   |format|,
%   |justification|,
%   |label|\-|font|,
%   |label|\-|format|,
%   |label|\-|separator|,
%   |list|\-|format|,
%   |margin|,
%   |parbox|,
%   |position|,
%   |single|\-|line|\-|check|,
%   |text|\-|font|,
%   |text|\-|format|, and
%   |width|.
%
% The starred variant only makes a difference when setting the default length
% of either |margin| or |width|, in this case it will use |\def| internally
% instead of |\edef| to store the value so the actual margin will be calculated
% when used (and not when set).
%
% \DescribeMacro{\SetCaptionFallback}
% The command
% \begin{quote}
%   |\SetCaptionFallback|\marg{option}\marg{fallback value}
% \end{quote}
% changes the fallback value for the option \meta{option}, for example:
% \begin{quote}
%   |\SetCaptionFallback{labelformat}{simple}|
% \end{quote}
% Currently only the pre-defined label format `original' uses a fallback
% value when |\fnum@|\meta{environment} is not defined for the current (floating)
% environment.
%
% \DescribeMacro{\AtCaptionPackage}
% The adaption file will be loaded when the \pkg{caption} kernel |caption3.sty|
% will be loaded so the given adaptions will be done for the \pkg{floatrow}
% and \pkg{subfig} packages even if the \pkg{caption} package isn't loaded.
%

% \section{Why this package is doomed to fail}
% \label{doomed_to_fail}
%
% \begin{enumerate}
% \item
% \LaTeX\ offers the internal command |\@make|\-|caption| as interface from \LaTeX\ kernel to document class.
% The standard \LaTeX\ document classes (|article|, |report|, and |book|) have implemented it this way:
% Measure the caption width (as single line), and if it fits output the measuring (horizontal) box centered,
% otherwise typeset it as simple paragraph.
%
% This itself is already a source for incompatibilites:
% Some packages use |\@make|\-|caption|, too (like the \pkg{listings} package) and expect the same behaviour
% although this was meant to be implemented in different ways in other document classes.
%
% Furthermore this way the caption content will be evaluated once if typeset as centered horizontal
% box but twice if typeset as paragraph, causing unwanted side-effects like double incremention of counters etc.
% And some content will be typeset differently dependant on which way it is typeset;
% for example a line break (|\\|) will be ignored in short, centered captions, but has the expected effect in
% long captions which are typeset as paragraph.
%
% The caption package in contrast always evaluates the caption content twice and has implemented the behaviour
% above in a different way, by applying additional options if it would fit into a single line.
% (This could be switched off by using the |singlelinecheck=| option.)
% This way the output is more consistent between short and long captions, but could cause unexpected behaviour
% in short captions (which would otherwise only appear in longer captions).
%
% \item
% Since the caption interface offered by \LaTeX\ is so limited many document classes have their
% own |\caption| code.
% Therefore the \pkg{caption} package needs to be adapted to these document classes or vice versa,
% which haven't been done (yet) for many document classes.
%
% \item
% For some options to work the \pkg{caption} package needs to typeset the caption content in a box.
% But \TeX\ boxes have only one baseline, and therefore either the vertical spacing above or below will be incorrect.
% There is a work-around macro called |\strut| to compensate for this (in an imperfect manner),
% and the caption package is using this by default, but it could cause unexpected behaviour in some use cases, too.
%
% \item
% |\@make|\-|caption| is only designed to work correctly in floating environments under certain conditions
% (usually prepared by the internal \LaTeX\ command |\@par|\-|box|\-|restore|)
% but some packages (like the \pkg{listing} package) use it in non-floating environments as well.
% Furthermore users could typeset listings in a \LaTeX\ list (like |itemize|) but |\@make|\-|caption| is not
% designed to give the expected outcome in LaTeX lists.
% (Sometimes the user is not even aware of using a \LaTeX\ list, for example the environment |adjust|\-|width|
% offered by the \pkg{changepage} package uses a LaTeX list internally to achieve its goal.)
%
% The caption package tries to detect such situations in order to handle them in a special way,
% but this will not always work.
%
% \item
% Many packages (|float|, |longtable|, |wrapfig|, ...) have their own caption code instead of using |\@make|\-|caption|.
% The caption package tries its best to patch these packages so the outcome is as expected,
% but in some cases the best the caption package could offer is an approximation of the original outcome.
%
% \item
% Some babel packages are patching caption related code, too. And they are doing it very inconsistently:
% Some packages always do all the patch work (although the user might typeset only a single paragraph in this language),
% some do all the patching when switching the language, and some do a mixture of both.
% Furthermore some of these packages are quite dirty and only work correctly when using one of the standard LaTeX document classes
% or very specific versions of the caption package.
%
% \item
% There are solutions out there which heavily depend on internals of the code of |\@make|\-|caption|.
% For example \url{https://comp.text.tex.narkive.com/XkJnYryR/figure-without-caption} re-defines
% |\fnum@figure| or |\fnum@table| to have an additional, unused argument:
% \begin{quote}
%   |\renewcommand\fnum@figure[1]{\ignorespaces}|
% \end{quote}
% This way the `:' which is hard coded into the standard implementation of |\@make|\-|caption|
% magically disappears, but only if the ':' is directly following the usage of the caption label!
% Of course the \pkg{caption} package (and other implementations of |\@make|\-|caption|)
% have a different behaviour and for this reason something other than `:' disappers,
% causing either an unexpected outcome or an incomprehensive error message.
% (Note: I detected this problem in 2020 and have added a work-around for this
% in the \pkg{caption} package, so the hack will work correctly.)
%
% BTW: This is a general problem when using \LaTeX.
% Since the regular user is unable to master \LaTeX\ and all its packages
% (due to different user interface designs per package, and due to a massive amount of package incompatiblities and/or package load-order problems),
% the user is forced to copy incomprehensibe code from internet sites into their documents.
% Especially he could not distinguish between ``good'' and ``bad'' packages
% (and while Linux distributions tend to include ``good'' applications only, TeX distributions like TeXlive tend to include every LaTeX package available on CTAN),
% he could not distinguish between ``good'' and ``bad'' code,
% and (at least in my experience) neither the voting of the answer nor the so called ``reputation'' of its creator is a suitable critria.
% Unfortunately it's often easier for an insider to write a quick \& dirty solution instead of a clean one.
% Back to the example above from 2005: The dirty solution is a one-liner, but the clean solution would be writing something like the \pkg{caption} package so the user could get rid of the hard coded `:' in a cleaner (and maintained) way.
% This problem is reinforced by the highly non-modular nature of the \LaTeX\ kernel itself.
% Outsourcing the `:' into a macro would have avoided this problem, but would have caused a memory increase by some bytes which were very rare in 1984.
% (This approach is contrary to current ``clean code'' standards set by Robert C. Martin and others.)
% And afterwards it was avoided to change internal \LaTeX\ code because so many packages and user documents already relied on the existing implementation,
% so every change for the better had the potential for a problem somewhere else.
% (And the vast majority of \LaTeX\ packages isn't maintained anymore and therefore would either stop working correctly or has to be maintained by the \LaTeX\ team itself to make it work again.)
%
% \item
% The LaTeX implementation for captions have an odd counter handling, i.e. not the figure or table environment increases the counter but |\caption| does.
% This is already a source of user confusion (since |\label| has to be placed after |\caption|, or sometimes only used inside |\caption| will give the correct outcome),
% and the confusion and user frustrations grows when using other environents which behaves differently (like |long|\-|table| offered by the \pkg{longtable} package),
% but this problem really shines brightly when using sub-captions additionally. (See section ``Numbering'' in the \pkg{subcaption} package documentation.)
% \end{enumerate}
%
% For all these reasons the amount of code which actually do typeset the caption is outweight by the amount of code which tries to herd cats by patching existing
% \LaTeX\ kernel, document class, or package code, and by implementing many work-arounds.
% (Any many of these patches and work-arounds have a bias to specific use case scenarios since they simply cannot be implemented in a complete compatible way.)
%
